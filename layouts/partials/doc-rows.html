Call it through partialCached keyed on type and version: once per type and version for the whole build, so nothing in here may depend on the current page. The link of the current page is marked by main.js. A version that has release notes links them from its number, as the component table of a platform release does. Nothing when the version has neither documentation nor release notes. */ -}} {{- $type := .type -}} {{- $version := .version -}} {{- $pages := index (partialCached "doc-pages.html" "doc-pages") (printf "%v|%v" $type $version) | default slice -}} {{- $docs := sort (where $pages "Params.type" "ne" "release notes") "Params.language" -}} {{- $notes := partial "notes-page.html" (dict "origin" $type "version" $version) -}} {{- if or $docs $notes }} {{ $type }} {{ with $notes }}{{ $version }}{{ else }}{{ $version }}{{ end }} {{- with $docs }}
    {{- range . }}
  • {{ .Params.language }}
    • {{- end }}
{{- end -}} {{- end }}
