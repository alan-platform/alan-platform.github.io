{{- /* Rows for one system type or platform component, (dict "type" T "version" V), selected by the front matter of the documentation pages. Call it through partialCached keyed on type and version: once per type and version for the whole build, so nothing in here may depend on the current page. The current page's row is highlighted by main.js. A version that has release notes links them from its number, as the component table of a platform release does. */ -}} {{- $type := .type -}} {{- $version := .version -}} {{- $pages := index (partialCached "doc-pages.html" "doc-pages") (printf "%v|%v" $type $version) | default slice -}} {{- $docs := sort (where $pages "Params.type" "ne" "release notes") "Params.language" -}} {{- $notes := partial "notes-page.html" (dict "origin" $type "version" $version) -}} {{- range $i, $doc := $docs }} {{ if eq $i 0 }}{{ $type }}{{ end }} {{ if eq $i 0 }}{{ with $notes }}{{ $version }}{{ else }}{{ $version }}{{ end }}{{ end }} {{ $doc.Params.language }} {{- else }} {{- with $notes }} {{ $type }} {{ $version }} {{- end }} {{- end }}
