{{- /* The keys of a map from the data files, in the order that hugo.toml names (params.order): (dict "map" M "kind" K), where K is "platform" (M is the platform object of a version in _data/versions.json) or "system types" (M is "system types" of _data/dist//versions.json). Hugo hands data over as maps, and a range over a map goes in sorted key order. The tables keep the order of the files: platform components as _data/versions.json lists them (model before interface), system types as the dist versions.json files list them (older versions are not alphabetical). Every file agrees with the lists of hugo.toml; a key that a list does not name, such as a component that a new version adds, follows in sorted order, so that nothing in the map is left out. */ -}} {{- $order := slice -}} {{- with site.Params.order -}} {{- if eq $.kind "platform" -}} {{- $order = .platform -}} {{- else if eq $.kind "system types" -}} {{- $order = index . "systemtypes" -}} {{- else -}} {{- errorf "ordered-keys: unknown kind %q, expected platform or system types" $.kind -}} {{- end -}} {{- else -}} {{- errorf "ordered-keys: [params.order] is missing from hugo.toml" -}} {{- end -}} {{- $keys := slice -}} {{- range $order -}} {{- if and (isset $.map .) (not (in $keys .)) -}} {{- $keys = $keys | append . -}} {{- end -}} {{- end -}} {{- range $key, $value := .map -}} {{- if not (in $keys $key) -}} {{- $keys = $keys | append $key -}} {{- end -}} {{- end -}} {{- return $keys -}}
