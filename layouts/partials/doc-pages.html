{{- /* The documentation pages by component version: a dict from "|" to the list of pages whose front matter has that origin and version. Versions are compared as text: a page with version: 101 belongs to the component version "101" of the data files. Call it through partialCached: it walks the pages once per build. The partial notes-page picks the release notes out of a list. */ -}} {{- $index := newScratch -}} {{- range $page := site.RegularPages -}} {{- with $page.Params.origin -}} {{- $index.Add (printf "%v|%v" . $page.Params.version) (slice $page) -}} {{- end -}} {{- end -}} {{- return $index.Values -}}
